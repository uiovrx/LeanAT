add_test(NAME leanat_original_conformance
  COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/run.py
    --root ${PROJECT_SOURCE_DIR} --output ${CMAKE_CURRENT_BINARY_DIR}/results.json
    ${LEANAT_CONFORMANCE_ARGS})
# The complete fresh E39 and E40 corpora exceed ten minutes; their individual
# subprocess timeouts remain enforced within this bounded aggregate allowance.
set_tests_properties(leanat_original_conformance PROPERTIES TIMEOUT 7200)
add_custom_target(conformance-report
  COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/run.py
    --root ${PROJECT_SOURCE_DIR} --output ${CMAKE_CURRENT_BINARY_DIR}/results.json
    ${LEANAT_CONFORMANCE_ARGS} --allow-incomplete
  DEPENDS leanat_conformance_native)
